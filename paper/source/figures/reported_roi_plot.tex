\newcommand{\roirow}[4]{%
  \pgfplotstablegetelem{#1}{reported_roi_signal}\of{\roidata}%
  \edef\roiraw{\pgfplotsretval}%
  \begingroup
    \pgfkeys{/pgf/fpu=true}%
    \pgfmathparse{\roiraw/1000000}%
    \pgfmathfloattofixed{\pgfmathresult}%
    \xdef\roivalue{\pgfmathresult}%
  \endgroup
  \begingroup
    \pgfkeys{/pgf/fpu=true}%
    \pgfmathparse{ln(\roiraw)/ln(10)}%
    \pgfmathfloattofixed{\pgfmathresult}%
    \xdef\roilog{\pgfmathresult}%
  \endgroup
  \pgfmathsetmacro{\bartop}{34+34*(\roilog-4)}%
  \filldraw[fill=#4,draw=#4!75!black,line width=.55pt]
    (#3-22,34) rectangle (#3+22,\bartop);
  \node[anchor=south,font=\footnotesize\bfseries,text=roiink] at (#3,\bartop+3)
    {\pgfmathprintnumber[sci,precision=3]{\roiraw}};
  \node[anchor=north,font=\footnotesize,text=roiink] at (#3,26) {#2};
}
